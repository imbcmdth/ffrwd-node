  -m still=still.wasm -filter_complex \
  '[v=0:v]still=shortest=1:longest=2:tolerance=2[stills=out0]' -bound \
  'still=[{"input":"v","streams":[{"rate":{"num":15,"den":1}}]}]' -map '[out0]' -f \
  ndjson stills.ndjson
